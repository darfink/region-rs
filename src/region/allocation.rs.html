7/// A handle to an owned region of memory.
8///
9/// This handle does not dereference to a slice, since the underlying memory may
10/// have been created with [`Protection::NONE`].
11pub struct Allocation {
12  base: *const (),
13  size: usize,
14}
15
16impl Allocation {
17  /// Returns a pointer to the allocation's base address.
18  ///
19  /// The address is always aligned to the operating system's page size.
20  #[inline(always)]
21  pub fn as_ptr<T>(&self) -> *const T {
22    self.base.cast()
23  }
24
25  /// Returns a mutable pointer to the allocation's base address.
26  #[inline(always)]
27  pub fn as_mut_ptr<T>(&mut self) -> *mut T {
28    self.base.cast_mut().cast()
29  }
30
31  /// Returns two raw pointers spanning the allocation's address space.
32  ///
33  /// The returned range is half-open, which means that the end pointer points
34  /// one past the last element of the allocation. This way, an empty allocation
35  /// is represented by two equal pointers, and the difference between the two
36  /// pointers represents the size of the allocation.
37  #[inline(always)]
38  pub fn as_ptr_range<T>(&self) -> Range<*const T> {
39    let range = self.as_range();
40    ptr::with_exposed_provenance::<T>(range.start)..ptr::with_exposed_provenance::<T>(range.end)
41  }
42
43  /// Returns two mutable raw pointers spanning the allocation's address space.
44  #[inline(always)]
45  pub fn as_mut_ptr_range<T>(&mut self) -> Range<*mut T> {
46    let range = self.as_range();
47    ptr::with_exposed_provenance_mut::<T>(range.start)
48      ..ptr::with_exposed_provenance_mut::<T>(range.end)
49  }
50
51  /// Returns a range spanning the allocation's address space.
52  #[inline(always)]
53  pub fn as_range(&self) -> Range<usize> {
54    let start = self.base.addr();
55    start..start.saturating_add(self.size)
56  }
57
58  /// Returns the size of the allocation in bytes.
59  ///
60  /// The size is always aligned to a multiple of the operating system's page
61  /// size.
62  #[inline(always)]
63  pub fn len(&self) -> usize {
64    self.size
65  }
66
67  /// Returns whether the allocation is empty or not.
68  #[inline(always)]
69  pub fn is_empty(&self) -> bool {
70    self.size == 0
71  }
72
73  /// Decomposes an `Allocation` into its raw components: `(pointer, length)`.
74  ///
75  /// After calling this function, the caller is responsible for the previously
76  /// managed allocation.
77  ///
78  /// For creating an `Allocation` from raw components, see [`Self::from_raw_parts`].
79  #[inline]
80  pub fn into_raw_parts<T>(self) -> (*mut T, usize) {
81    let mut this = ManuallyDrop::new(self);
82    (this.as_mut_ptr(), this.len())
83  }
84
85  /// Creates a `Allocation` directly from a pointer, and a length.
86  ///
87  /// For decomposing an `Allocation` into raw components, see
88  /// [`Self::into_raw_parts`].
89  ///
90  /// # Safety
91  ///
92  /// This is highly unsafe because given `ptr` and `length` could not
93  /// be checked as valid allocation, and the caller should guarantee
94  /// that they are valid parts.
95  #[inline(always)]
96  pub unsafe fn from_raw_parts<T>(ptr: *mut T, length: usize) -> Self {
97    Self {
98      base: ptr.cast(),
99      size: length,
100    }
101  }
102}
103
104impl Drop for Allocation {
105  #[inline]
106  fn drop(&mut self) {
107    let result = unsafe { os::free(self.base, self.size) };
108    debug_assert!(result.is_ok(), "freeing region: {:?}", result);
109  }
110}

112/// Allocates one or more pages of memory, with a defined protection.
113///
114/// This function provides a very simple interface for allocating anonymous
115/// virtual pages. The allocation address will be decided by the operating
116/// system.
117///
118/// # Parameters
119///
120/// - The size may not be zero.
121/// - The size is rounded up to the closest page boundary.
122///
123/// # Errors
124///
125/// - If an interaction with the underlying operating system fails, an error
126///   will be returned.
127/// - If size is zero, [`Error::InvalidParameter`] will be returned.
128///
129/// # OS-Specific Behavior
130///
131/// On NetBSD pages will be allocated without PaX memory protection restrictions
132/// (i.e. pages will be allowed to be modified to any combination of `RWX`).
133///
134/// On Windows, allocating with [`Protection::NONE`] reserves address space
135/// without committing physical pages.
136///
137/// # Examples
138///
139/// ```
140/// # fn main() -> region::Result<()> {
141/// # if cfg!(any(target_arch = "x86", target_arch = "x86_64"))
142/// #   && !cfg!(any(target_os = "openbsd", target_os = "netbsd")) {
143/// use region::Protection;
144/// let ret5 = [0xB8, 0x05, 0x00, 0x00, 0x00, 0xC3u8];
145///
146/// let memory = region::alloc(100, Protection::READ_WRITE_EXECUTE)?;
147/// let slice = unsafe {
148///   core::slice::from_raw_parts_mut(memory.as_ptr::<u8>().cast_mut(), memory.len())
149/// };
150///
151/// slice[..6].copy_from_slice(&ret5);
152/// let x: extern "C" fn() -> i32 = unsafe { core::mem::transmute(slice.as_ptr()) };
153///
154/// assert_eq!(x(), 5);
155/// # }
156/// # Ok(())
157/// # }
158/// ```
159#[inline]
160pub fn alloc(size: usize, protection: Protection) -> Result<Allocation> {
161  if size == 0 {
162    return Err(Error::InvalidParameter("size"));
163  }
164
165  let size = page::ceil(ptr::without_provenance::<()>(size)).addr();
166
167  unsafe {
168    let base = os::alloc(ptr::null::<()>(), size, protection)?;
169    Ok(Allocation { base, size })
170  }
171}

173/// Allocates one or more pages of memory, at a specific address, with a defined
174/// protection.
175///
176/// The returned memory allocation is not guaranteed to reside at the provided
177/// address. E.g. on Windows, new allocations that do not reside within already
178/// reserved memory, are aligned to the operating system's allocation
179/// granularity (most commonly 64KB).
180///
181/// # Implementation
182///
183/// This function is implemented using `VirtualAlloc` on Windows, and `mmap`
184/// with `MAP_FIXED` on POSIX.
185///
186/// # Parameters
187///
188/// - The address is rounded down to the closest page boundary.
189/// - The size may not be zero.
190/// - The size is rounded up to the closest page boundary, relative to the
191///   address.
192///
193/// # Errors
194///
195/// - If an interaction with the underlying operating system fails, an error
196///   will be returned.
197/// - If size is zero, [`Error::InvalidParameter`] will be returned.
198#[inline]
199pub fn alloc_at<T>(address: *const T, size: usize, protection: Protection) -> Result<Allocation> {
200  let (address, size) = util::round_to_page_boundaries(address, size)?;
201
202  unsafe {
203    let base = os::alloc(address.cast(), size, protection)?;
204    Ok(Allocation { base, size })
205  }
206}

227  #[test]
228  fn alloc_obtains_correct_properties() -> Result<()> {
229    let memory = alloc(1, Protection::READ_WRITE)?;
230
231    let region = crate::query(memory.as_ptr::<()>())?;
232    assert_eq!(region.protection(), Protection::READ_WRITE);
233    assert!(region.len() >= memory.len());
234    assert!(!region.is_guarded());
235    assert!(!region.is_shared());
236    assert!(region.is_committed());
237
238    Ok(())
239  }

241  #[test]
242  #[cfg(not(target_os = "netbsd"))]
243  fn alloc_frees_memory_when_dropped() -> Result<()> {
244    // Designing these tests can be quite tricky sometimes. When a page is
245    // allocated and then released, a subsequent `query` may allocate memory in
246    // the same location that has just been freed. For instance, NetBSD's
247    // kinfo_getvmmap uses `mmap` internally, which can lead to potentially
248    // confusing outcomes. Retain several buffer regions so any memory allocated
249    // indirectly through `query` occupies a separate location in memory.
250    let buffers = (0..8)
251      .map(|_| alloc(1, Protection::READ_WRITE))
252      .collect::<Result<alloc::vec::Vec<_>>>()?;
253    let start = alloc(1, Protection::READ_WRITE)?;
254
255    let base = start.as_ptr::<()>();
256    drop(start);
257
258    let query = crate::query(base);
259    assert!(
260      matches!(query, Err(Error::UnmappedRegion)),
261      "expected unmapped region after free, got {query:?}; retained {} buffers",
262      buffers.len()
263    );
264    Ok(())
265  }
266
267  #[test]
268  fn alloc_can_allocate_unused_region() -> Result<()> {
269    let base = alloc(1, Protection::NONE)?.as_ptr::<()>();
270    let memory = alloc_at(base, 1, Protection::READ_WRITE)?;
271    assert_eq!(memory.as_ptr(), base);
272    Ok(())
273  }
