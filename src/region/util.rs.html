2use core::ptr;

4/// Validates & rounds an address-size pair to their respective page boundary.
5pub fn round_to_page_boundaries<T>(address: *const T, size: usize) -> Result<(*const T, usize)> {
6  if size == 0 {
7    return Err(Error::InvalidParameter("size"));
8  }
9
10  let size = address.addr() % page::size() + size;
11  let size = page::ceil(ptr::with_exposed_provenance::<T>(size)).addr();
12  Ok((page::floor(address), size))
13}

16mod tests {

19  #[test]
20  fn round_to_page_boundaries_works() -> Result<()> {
21    let pz = page::size();
22    let values = &[
23      ((1, pz), (0, pz * 2)),
24      ((0, pz - 1), (0, pz)),
25      ((0, pz + 1), (0, pz * 2)),
26      ((pz - 1, 1), (0, pz)),
27      ((pz + 1, pz), (pz, pz * 2)),
28      ((pz, pz), (pz, pz)),
29    ];
30
31    for ((before_address, before_size), (after_address, after_size)) in values {
32      let (address, size) = round_to_page_boundaries(
33        ptr::with_exposed_provenance::<()>(*before_address),
34        *before_size,
35      )?;
36      assert_eq!(
37        (address, size),
38        (
39          ptr::with_exposed_provenance::<()>(*after_address),
40          *after_size
41        )
42      );
43    }
44    Ok(())
45  }
