5/// Changes the memory protection of one or more pages.
6///
7/// The address range may overlap one or more pages, and if so, all pages
8/// spanning the range will be modified. The previous protection flags are not
9/// preserved (if you desire to preserve the protection flags, use
10/// [`protect_with_handle`]).
11///
12/// # Parameters
13///
14/// - The range is `[address, address + size)`
15/// - The address is rounded down to the closest page boundary.
16/// - The size may not be zero.
17/// - The size is rounded up to the closest page boundary, relative to the
18///   address.
19///
20/// # Errors
21///
22/// - If an interaction with the underlying operating system fails, an error
23///   will be returned.
24/// - If size is zero,
25///   [`Error::InvalidParameter`](crate::Error::InvalidParameter) will be
26///   returned.
27///
28/// # Safety
29///
30/// This function can violate memory safety in a myriad of ways. Read-only memory
31/// can become writable, the executable properties of code segments can be
32/// removed, etc.
33///
34/// # Examples
35///
36/// - Make an array of x86 assembly instructions executable.
37///
38/// ```
39/// # fn main() -> region::Result<()> {
40/// # if cfg!(any(target_arch = "x86", target_arch = "x86_64"))
41/// #   && !cfg!(any(target_os = "openbsd", target_os = "netbsd")) {
42/// use region::Protection;
43/// let ret5 = [0xB8, 0x05, 0x00, 0x00, 0x00, 0xC3u8];
44///
45/// let x: extern "C" fn() -> i32 = unsafe {
46///   region::protect(ret5.as_ptr(), ret5.len(), region::Protection::READ_WRITE_EXECUTE)?;
47///   core::mem::transmute(ret5.as_ptr())
48/// };
49///
50/// assert_eq!(x(), 5);
51/// # }
52/// # Ok(())
53/// # }
54/// ```
55#[inline]
56pub unsafe fn protect<T>(address: *const T, size: usize, protection: Protection) -> Result<()> {
57  let (address, size) = util::round_to_page_boundaries(address, size)?;
58  unsafe { os::protect(address.cast(), size, protection) }
59}

61/// Temporarily changes the memory protection of one or more pages.
62///
63/// The address range may overlap one or more pages, and if so, all pages within
64/// the range will be modified. The protection flag for each page will be reset
65/// once the handle is dropped. To conditionally prevent a reset, use
66/// [`core::mem::forget`].
67///
68/// This function uses [`query_range`](crate::query_range) internally and is
69/// therefore less performant than [`protect`]. Use this function only if you
70/// need to reapply the memory protection flags of one or more regions after
71/// operations.
72///
73/// # Guard
74///
75/// Remember not to conflate the *black hole* syntax with the ignored, but
76/// unused, variable syntax. Otherwise the [`ProtectGuard`] instantly resets the
77/// protection flags of all pages.
78///
79/// ```ignore
80/// let _ = protect_with_handle(...);      // Pages are instantly reset
81/// let _guard = protect_with_handle(...); // Pages are reset once `_guard` is dropped.
82/// ```
83///
84/// # Parameters
85///
86/// - The range is `[address, address + size)`
87/// - The address is rounded down to the closest page boundary.
88/// - The size may not be zero.
89/// - The size is rounded up to the closest page boundary, relative to the
90///   address.
91///
92/// # Errors
93///
94/// - If an interaction with the underlying operating system fails, an error
95///   will be returned.
96/// - If size is zero,
97///   [`Error::InvalidParameter`](crate::Error::InvalidParameter) will be
98///   returned.
99///
100/// # Safety
101///
102/// See [protect].
103#[allow(clippy::missing_inline_in_public_items)]
104pub unsafe fn protect_with_handle<T>(
105  address: *const T,
106  size: usize,
107  protection: Protection,
108) -> Result<ProtectGuard> {
109  let (address, size) = util::round_to_page_boundaries(address, size)?;
110
111  // Preserve the current regions' flags
112  let mut regions = QueryIter::new(address, size)?.collect::<Result<Vec<_>>>()?;
113
114  // Apply the desired protection flags
115  unsafe { protect(address, size, protection)? };
116
117  if let Some(region) = regions.first_mut() {
118    // Offset the lower region to the smallest page boundary
119    region.base = address.cast();
120    region.size -= address.addr().saturating_sub(region.as_range().start);
121  }
122
123  if let Some(region) = regions.last_mut() {
124    // Truncate the upper region to the smallest page boundary
125    let protect_end = address.addr().saturating_add(size);
126    region.size -= region.as_range().end.saturating_sub(protect_end);
127  }
128
129  Ok(ProtectGuard::new(regions))
130}

132/// A RAII implementation of a scoped protection guard.
133///
134/// When this structure is dropped (falls out of scope), the memory regions'
135/// protection will be reset.
136#[must_use]
137pub struct ProtectGuard {
138  regions: Vec<Region>,
139}
140
141impl ProtectGuard {
142  #[inline(always)]
143  fn new(regions: Vec<Region>) -> Self {
144    Self { regions }
145  }
146}
147
148impl Drop for ProtectGuard {
149  #[inline]
150  fn drop(&mut self) {
151    let result = self
152      .regions
153      .iter()
154      .try_for_each(|region| unsafe { protect(region.base, region.size, region.protection) });
155    debug_assert!(result.is_ok(), "restoring region protection: {:?}", result);
156  }
157}

160unsafe impl Sync for ProtectGuard {}

190  #[test]
191  fn protect_updates_both_pages_for_straddling_range() -> Result<()> {
192    let pz = page::size();
193
194    // Create a page boundary with different protection flags in the upper and
195    // lower span, so the intermediate region sizes are fixed to one page.
196    let map = alloc_pages(&[
197      Protection::READ,
198      Protection::READ_EXECUTE,
199      Protection::READ_WRITE,
200      Protection::READ,
201    ]);
202
203    let exec_page = unsafe { map.as_ptr().add(pz) };
204    let exec_page_end = unsafe { exec_page.add(pz - 1) };
205
206    // Change the protection over two page boundaries
207    unsafe {
208      protect(exec_page_end, 2, Protection::NONE)?;
209    }
210
211    // Query the two inner pages
212    let result = query_range(exec_page, pz * 2)?.collect::<Result<Vec<_>>>()?;
213
214    // On some OSs the pages are merged into one region
215    assert!(matches!(result.len(), 1 | 2));
216    assert_eq!(result.iter().map(Region::len).sum::<usize>(), pz * 2);
217    assert_eq!(result[0].protection(), Protection::NONE);
218    Ok(())
219  }

222  fn protect_has_inclusive_lower_and_exclusive_upper_bound() -> Result<()> {
223    let map = alloc_pages(&[
224      Protection::READ_WRITE,
225      Protection::READ,
226      Protection::READ_WRITE,
227      Protection::READ,
228    ]);
229
230    // Alter the protection of the second page
231    let second_page = unsafe { map.as_ptr().add(page::size()) };
232    unsafe {
233      let second_page_end = second_page.offset(page::size() as isize - 1);
234      protect(second_page_end, 1, Protection::NONE)?;
235    }
236
237    let regions = query_range(map.as_ptr(), page::size() * 3)?.collect::<Result<Vec<_>>>()?;
238    assert_eq!(regions.len(), 3);
239    assert_eq!(regions[0].protection(), Protection::READ_WRITE);
240    assert_eq!(regions[1].protection(), Protection::NONE);
241    assert_eq!(regions[2].protection(), Protection::READ_WRITE);
242
243    // Alter the protection of '2nd_page_start .. 2nd_page_end + 1'
244    unsafe {
245      protect(second_page, page::size() + 1, Protection::READ_EXECUTE)?;
246    }
247
248    let regions = query_range(map.as_ptr(), page::size() * 3)?.collect::<Result<Vec<_>>>()?;
249    assert!(regions.len() >= 2);
250    assert_eq!(regions[0].protection(), Protection::READ_WRITE);
251    assert_eq!(regions[1].protection(), Protection::READ_EXECUTE);
252    assert!(regions[1].len() >= page::size());

270  #[test]
271  fn protect_with_handle_only_alters_protection_of_affected_pages() -> Result<()> {
272    let pages = [
273      Protection::READ_WRITE,
274      Protection::READ,
275      Protection::READ_WRITE,
276      Protection::READ_EXECUTE,
277      Protection::NONE,
278    ];
279    let map = alloc_pages(&pages);
280
281    let second_page = unsafe { map.as_ptr().add(page::size()) };
282    let region_size = page::size() * 3;
283
284    unsafe {
285      let _handle = protect_with_handle(second_page, region_size, Protection::NONE)?;
286      let region = query(second_page)?;
287
288      assert_eq!(region.protection(), Protection::NONE);
289      // Some platforms may merge adjacent identically protected pages, so the
290      // returned region base can precede `second_page` as long as it contains it.
291      assert!(region.as_range().contains(&second_page.addr()));
292    }
293
294    let regions =
295      query_range(map.as_ptr(), page::size() * pages.len())?.collect::<Result<Vec<_>>>()?;
296
297    // Region descriptors may be merged or split depending on the OS query
298    // backend; compare by page rather than assuming a 1:1 region count/base.
299    assert!(!regions.is_empty());
300    for (index, expected) in pages.iter().enumerate() {
301      let address = unsafe { map.as_ptr().add(page::size() * index) };
302      let region = regions
303        .iter()
304        .find(|region| region.as_range().contains(&address.addr()))
305        .expect("expected region covering page");
306      assert_eq!(region.protection(), *expected);
307    }
308
309    Ok(())
310  }
